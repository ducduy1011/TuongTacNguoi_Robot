cmake_minimum_required(VERSION 3.16)
project(btth4_slam_solver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(Ceres REQUIRED)
find_package(Boost REQUIRED COMPONENTS system serialization filesystem thread)
find_library(KARTO_LIBRARY kartoSlamToolbox
  HINTS "${slam_toolbox_DIR}/../../../lib" REQUIRED)
if(NOT EXISTS "${SOLVER_SOURCE}/solvers/ceres_solver.cpp")
  message(FATAL_ERROR "Run python3 tools/prepare_runtime.py to download the pinned official solver.")
endif()
add_library(ceres_solver_plugin SHARED "${SOLVER_SOURCE}/solvers/ceres_solver.cpp")
target_include_directories(ceres_solver_plugin PRIVATE
  "${SOLVER_SOURCE}" ${slam_toolbox_INCLUDE_DIRS} ${CERES_INCLUDE_DIRS})
ament_target_dependencies(ceres_solver_plugin rclcpp pluginlib std_srvs
  tf2_ros tf2_msgs nav_msgs sensor_msgs visualization_msgs)
target_link_libraries(ceres_solver_plugin ${CERES_LIBRARIES} ${Boost_LIBRARIES} ${KARTO_LIBRARY})
install(TARGETS ceres_solver_plugin LIBRARY DESTINATION lib)

cmake_minimum_required(VERSION 3.8)
project(ur3_llm_control_week3)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/ExecuteSkill.srv" "srv/GetSceneState.srv")
install(PROGRAMS scripts/llm_task.py scripts/user_command_cli.py scripts/scene_perception.py scripts/scene_visualizer.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config urdf worlds DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
find_package(rclcpp REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_ros_move_group REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(skill_server src/skill_server.cpp)
ament_target_dependencies(skill_server rclcpp rclcpp_action control_msgs std_srvs std_msgs ament_index_cpp moveit_ros_planning_interface shape_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(skill_server "${typesupport}" yaml-cpp)
install(TARGETS skill_server DESTINATION lib/${PROJECT_NAME})
add_library(moveit_shutdown SHARED src/moveit_shutdown.cpp)
ament_target_dependencies(moveit_shutdown rclcpp moveit_ros_planning_interface)
target_link_libraries(moveit_shutdown ${CMAKE_DL_LIBS})
install(TARGETS moveit_shutdown LIBRARY DESTINATION lib)
find_package(ignition-gazebo6 REQUIRED)
add_library(physical_grasp_system SHARED src/physical_grasp_system.cpp)
ament_target_dependencies(physical_grasp_system rclcpp std_srvs)
target_link_libraries(physical_grasp_system ignition-gazebo6::ignition-gazebo6 ${CMAKE_DL_LIBS})
install(TARGETS physical_grasp_system LIBRARY DESTINATION lib)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
